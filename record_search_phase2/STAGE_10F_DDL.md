# 10f — `ddl`: the column exists, and the migration rebuilds it in Go

Repo `webtyp/ddl`. Decisions: RECORD_SEARCH D1 (column generated by ddl), D2 (full rebuild only when the column is
new or the list of fields that enter changed; SQLite cannot strip accents, so the rebuild runs in Go).
Compiled and tested green on 2026-11-16.

What it does in `Sync(m)`:
1. The schema Sync reconciles is `m.Schema()` plus `text_search` (TEXT, nullable) when any field enters it, so the
   column is added (`OpAddColumn`) and the safe-drop never drops it.
2. After the columns, `syncTextSearch(m)`: reads `text_search_state.fields` for this table; when missing or different
   from today's list (`"rut,name,phone,email"`), rewrites `text_search` of every row (`storage.RewriteTextSearch`,
   `fields` nil) and stores the new list. Same transaction as the rest of Sync when the connection has one.
3. A connection that cannot read rows (DDL-only `Execer`) on a table that needs the rebuild → error
   `errTextSearchNoReads` (never a silent empty column). `ddl.Capture`'s recorder is marked `recordsOnly` and skips it.

## 1. New file `text_search.go`

```go
package ddl

import (
	"webtyp.com/fmt"
	"webtyp.com/model"
	"webtyp.com/storage"
)

const (
	// textSearchStateTable is where ddl remembers, per table, the fields its text_search was built from, so a
	// migration rebuilds the column only when it is new or that list changed (RECORD_SEARCH D2).
	textSearchStateTable = "text_search_state"
	textSearchStateKey   = "table_name"
	textSearchStateValue = "fields"

	errTextSearchNoReads = "ddl: text_search of this table must be rebuilt, and the connection cannot read rows; run the migration with a storage.Conn:"
)

var textSearchStateFields = []model.Field{
	{Name: textSearchStateKey, Type: model.Text(), DB: &model.FieldDB{PK: true}},
	{Name: textSearchStateValue, Type: model.Text(), NotNull: true},
}

// recordsOnly is implemented by the Capture recorder: a connection that touches no database has no rows to rebuild.
type recordsOnly interface{ recordsOnly() }

// withTextSearchColumn is the schema Sync reconciles: the model's fields plus text_search when any field enters it.
func withTextSearchColumn(schema []model.Field) []model.Field {
	if len(model.TextSearchFields(schema)) == 0 {
		return schema
	}
	out := make([]model.Field, len(schema), len(schema)+1)
	copy(out, schema)
	return append(out, model.Field{Name: model.FieldTextSearch, Type: model.Text(), OmitEmpty: true})
}

// textSearchFieldList is the list text_search is built from, as ddl stores it: field names joined by ",".
func textSearchFieldList(schema []model.Field) string {
	idx := model.TextSearchFields(schema)
	names := make([]string, len(idx))
	for j, i := range idx {
		names[j] = schema[i].Name
	}
	return fmt.JoinSlice(names, ",")
}

// syncTextSearch rebuilds text_search of every row of m's table when the column is new or the fields that enter
// it changed since the last migration, and records the new list. A table without such fields does nothing.
func (d *DB) syncTextSearch(m model.Model) error {
	want := textSearchFieldList(m.Schema())
	if want == "" {
		return nil
	}
	if _, dry := d.conn.(recordsOnly); dry {
		return nil
	}
	conn, ok := d.conn.(storage.Conn)
	if !ok {
		return fmt.Err(errTextSearchNoReads, m.ModelName())
	}
	state := &schemaModel{name: textSearchStateTable, fields: textSearchStateFields}
	if err := d.createTable(state); err != nil {
		return err
	}
	byTable := []storage.Condition{storage.Eq(textSearchStateKey, m.ModelName())}
	plan, err := conn.Compile(storage.Query{Action: storage.ActionReadAll, Table: textSearchStateTable, Columns: []string{textSearchStateValue}, Conditions: byTable, Limit: 1}, state)
	if err != nil {
		return err
	}
	rows, err := conn.Query(plan.Query, plan.Args...)
	if err != nil {
		return err
	}
	var stored string
	found := rows.Next()
	if found {
		if err := rows.Scan(&stored); err != nil {
			rows.Close()
			return err
		}
	}
	rows.Close()
	if found && stored == want {
		return nil
	}
	if err := storage.RewriteTextSearch(conn, m, nil, nil); err != nil {
		return err
	}
	q := storage.Query{Action: storage.ActionCreate, Table: textSearchStateTable, Columns: []string{textSearchStateKey, textSearchStateValue}, Values: []any{m.ModelName(), want}}
	if found {
		q = storage.Query{Action: storage.ActionUpdate, Table: textSearchStateTable, Columns: []string{textSearchStateValue}, Values: []any{want}, Conditions: byTable}
	}
	plan, err = conn.Compile(q, state)
	if err != nil {
		return err
	}
	_, err = conn.Exec(plan.Query, plan.Args...)
	return err
}
```

## 3. Tests — new file `tests/text_search_test.go` (package `ddl_test`; uses `recordingCompiler`, `testModel`, `execOnly` from `tests/setup_test.go`)

```go
package ddl_test

import (
	"testing"

	"webtyp.com/ddl"
	"webtyp.com/model"
	"webtyp.com/storage"
	"webtyp.com/storage/mem"
)

// searchKind is a text kind that enters text_search (webtyp.com/input does this for Text, Phone, Email, Rut).
type searchKind struct{ model.Kind }

func (searchKind) SearchableText() {}

// memDDL is storage/mem with DDL accepted as a no-op (mem creates a table on its first insert), so Sync's
// DML half — the text_search rebuild — runs for real.
type memDDL struct{ storage.Conn }

func (c memDDL) Exec(query string, args ...any) (storage.Result, error) {
	if query == "ddl" { // what recordingCompiler compiles every DDL statement to
		return storage.Affected(0), nil
	}
	return c.Conn.Exec(query, args...)
}

func patientFields(withPhone bool) []model.Field {
	f := []model.Field{
		{Name: "id", Type: model.Text(), DB: &model.FieldDB{PK: true}},
		{Name: "name", Type: searchKind{model.Text()}},
		{Name: "phone", Type: model.Text()},
	}
	if withPhone {
		f[2].Type = searchKind{model.Text()}
	}
	return f
}

// insertPatient writes a row as an older version of the app did: without text_search.
func insertPatient(t *testing.T, conn storage.Conn, id, name, phone string) {
	t.Helper()
	m := &testModel{table: "patient", fields: patientFields(false)}
	plan, err := conn.Compile(storage.Query{Action: storage.ActionCreate, Table: "patient", Columns: []string{"id", "name", "phone"}, Values: []any{id, name, phone}}, m)
	if err != nil {
		t.Fatal(err)
	}
	if _, err := conn.Exec(plan.Query, plan.Args...); err != nil {
		t.Fatal(err)
	}
}

func textSearchOfPatient(t *testing.T, conn storage.Conn, id string) string {
	t.Helper()
	m := &testModel{table: "patient", fields: patientFields(false)}
	plan, err := conn.Compile(storage.Query{Action: storage.ActionReadOne, Table: "patient", Columns: []string{model.FieldTextSearch}, Conditions: []storage.Condition{storage.Eq("id", id)}, Limit: 1}, m)
	if err != nil {
		t.Fatal(err)
	}
	var got string
	if err := conn.QueryRow(plan.Query, plan.Args...).Scan(&got); err != nil {
		t.Fatal(err)
	}
	return got
}

func TestTextSearch_SyncRebuildsExistingRows(t *testing.T) {
	conn := memDDL{mem.New()}
	insertPatient(t, conn, "p1", "Lucía Pérez", "+56911111111")
	if err := ddl.New(conn, &recordingCompiler{}).Sync(&testModel{table: "patient", fields: patientFields(false)}); err != nil {
		t.Fatal(err)
	}
	if got := textSearchOfPatient(t, conn, "p1"); got != "lucia perez" {
		t.Errorf("text_search = %q, want %q", got, "lucia perez")
	}
}

func TestTextSearch_SyncAddsTheColumn(t *testing.T) {
	compiler := &recordingCompiler{}
	if err := ddl.New(memDDL{mem.New()}, compiler).Sync(&testModel{table: "patient", fields: patientFields(false)}); err != nil {
		t.Fatal(err)
	}
	for _, s := range compiler.stmts {
		if s.Op == ddl.OpAddColumn && s.Column.Name == model.FieldTextSearch {
			return
		}
	}
	t.Errorf("no ADD COLUMN %s in %+v", model.FieldTextSearch, compiler.stmts)
}

func TestTextSearch_SyncRebuildsWhenTheFieldsChange(t *testing.T) {
	conn := memDDL{mem.New()}
	insertPatient(t, conn, "p1", "Ana", "+56911111111")
	db := ddl.New(conn, &recordingCompiler{})
	if err := db.Sync(&testModel{table: "patient", fields: patientFields(false)}); err != nil {
		t.Fatal(err)
	}
	if err := db.Sync(&testModel{table: "patient", fields: patientFields(true)}); err != nil {
		t.Fatal(err)
	}
	if got := textSearchOfPatient(t, conn, "p1"); got != "ana 56911111111" {
		t.Errorf("text_search = %q, want %q", got, "ana 56911111111")
	}
}

func TestTextSearch_SyncDoesNotRebuildWhenNothingChanged(t *testing.T) {
	conn := memDDL{mem.New()}
	insertPatient(t, conn, "p1", "Ana", "")
	db := ddl.New(conn, &recordingCompiler{})
	m := &testModel{table: "patient", fields: patientFields(false)}
	if err := db.Sync(m); err != nil {
		t.Fatal(err)
	}
	// a write that bypassed the framework leaves a value Sync must not overwrite when the fields did not change
	plan, _ := conn.Compile(storage.Query{Action: storage.ActionUpdate, Table: "patient", Columns: []string{model.FieldTextSearch}, Values: []any{"kept"}, Conditions: []storage.Condition{storage.Eq("id", "p1")}}, m)
	if _, err := conn.Exec(plan.Query, plan.Args...); err != nil {
		t.Fatal(err)
	}
	if err := db.Sync(m); err != nil {
		t.Fatal(err)
	}
	if got := textSearchOfPatient(t, conn, "p1"); got != "kept" {
		t.Errorf("text_search = %q, want kept (no rebuild)", got)
	}
}

func TestTextSearch_SyncWithoutReadsFails(t *testing.T) {
	err := ddl.New(execOnly{}, &recordingCompiler{}).Sync(&testModel{table: "patient", fields: patientFields(false)})
	want := "ddl: text_search of this table must be rebuilt, and the connection cannot read rows; run the migration with a storage.Conn: patient"
	if err == nil || err.Error() != want {
		t.Errorf("err = %v, want %q", err, want)
	}
}

func TestTextSearch_CaptureRecordsOnlyTheModelTable(t *testing.T) {
	m := &testModel{table: "patient", fields: patientFields(false)}
	tables, err := ddl.Capture([]ddl.Step{{Name: "patient_directory", Run: func(conn ddl.Execer, c ddl.Compiler) error {
		return ddl.New(conn, c).Sync(m)
	}}})
	if err != nil {
		t.Fatal(err)
	}
	if len(tables) != 1 || tables[0].Model.ModelName() != "patient" {
		t.Errorf("tables = %+v, want only patient", tables)
	}
}
```
